def emit(value):
    print(json.dumps(value, ensure_ascii=False, default=str), file=OUT, flush=True)

def save(name, value):
    write_json(BASE / name, json.loads(json.dumps(value, ensure_ascii=False, default=str)))

def inventory(root):
    return {p.relative_to(root).as_posix(): sha256_file(p)
            for p in sorted(root.rglob('*')) if p.is_file() and not p.is_symlink()}

def protected():
    paths = {
        'root_agents': ROOT.parent / 'AGENTS.md',
        'v1_latest': ROOT / 'output/latest_run.json',
        'v1_ledger': ROOT / 'output/run_ledger.ndjson',
        'v2_latest': FORMAL / 'latest_run.json',
        'v2_ledger': FORMAL / 'run_ledger.ndjson',
        'run004': ROOT / 'output/runs/SWIVD-RUN-20260828-004/manifest.json',
        'run002': FORMAL / 'runs' / OLD_ID / 'manifest.json',
    }
    return {k: sha256_file(v) for k, v in paths.items()}

def safe_error(exc):
    code = getattr(exc, 'code', None)
    result = {'exception_type': type(exc).__name__, 'message_withheld': True}
    if isinstance(code, int) or isinstance(code, str) and re.fullmatch('[A-Z0-9_]{1,100}', code):
        result['code'] = code
    return result

def subset(manifest):
    keys = ('run_id', 'as_of', 'purpose', 'schema_version', 'provider_kind',
            'live_validation_state', 'artifact_publish_state', 'research_grade',
            'decision_eligible', 'production_approved', 'target_selection')
    return {k: manifest[k] for k in keys if k in manifest}

last = [None, 0.0]
def progress(phase, completed, total, item):
    now = time.monotonic()
    if phase != last[0] or now-last[1] >= 15:
        emit({'stage': phase, 'completed': completed, 'total': total})
        last[:] = [phase, now]

    state = {'decision_id': 'GOV-20260906-003', 'data_scope': 'SYSTEM_TEMP_ONLY',
             'source_modified': True, 'formal_published': False, 'checks': []}

            for purpose, as_of in (('UPDATE_LATEST', None), ('MATERIALIZE_DATE', '20260901')):

                emit({'operation': purpose, 'starting': True})

                    manifest = execute_locked(spec_path=ROOT / 'PROJECT_SPEC_V4.json', data_dir=DATA,

                assert manifest['provider_kind'] == 'LIVE_SECURE_TUSHARE'
                run = DATA / 'runs' / manifest['run_id']
